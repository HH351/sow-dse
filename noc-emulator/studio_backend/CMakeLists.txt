cmake_minimum_required(VERSION 3.20)
project(noc_studio LANGUAGES CXX)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
add_library(noc_studio SHARED src/studio_backend.cpp src/device.cpp)
target_compile_features(noc_studio PRIVATE cxx_std_20)
target_include_directories(noc_studio PUBLIC include)
target_link_libraries(noc_studio PRIVATE Threads::Threads nlohmann_json::nlohmann_json)
target_compile_options(noc_studio PRIVATE -Wall -Wextra -Wpedantic)
target_link_options(noc_studio PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
set_target_properties(noc_studio PROPERTIES VERSION 1.0.0 SOVERSION 1
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
